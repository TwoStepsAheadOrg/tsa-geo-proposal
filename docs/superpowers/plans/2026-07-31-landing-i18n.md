```js
export const LANGS = ["ko", "en", "ja"];

export function langFromPath(pathname) {
  const segment = String(pathname || "").split("/")[1];
  return segment === "en" || segment === "ja" ? segment : "ko";
}

export function pageFromPath(pathname) {
  const rest = String(pathname || "").replace(/^\/(en|ja)(?=\/|$)/, "");
  return rest.startsWith("/consulting") ? "consulting" : "plurank";
}

export function pathFor(lang, page) {
  const prefix = lang === "ko" ? "" : `/${lang}`;
  if (page === "consulting") return `${prefix}/consulting`;
  return prefix || "/";
}
```

```jsx
export function Experience() {
  const [page, setPage] = useState(() => pageFromPath(window.location.pathname));
  const [language, setLanguage] = useState(() => langFromPath(window.location.pathname));
  const [contactOpen, setContactOpen] = useState(false);
  const copy = COPY[language];

  useEffect(() => {
    const onPopState = () => {
      setPage(pageFromPath(window.location.pathname));
      setLanguage(langFromPath(window.location.pathname));
      window.scrollTo({ top: 0 });
    };
    window.addEventListener("popstate", onPopState);
    return () => window.removeEventListener("popstate", onPopState);
  }, []);

  useEffect(() => {
    document.documentElement.lang = language;
    document.title = documentTitle(language, page);
  }, [language, page]);

  const navigate = (target, anchor = null) => {
    const pathname = pathFor(language, target);
    const destination = anchor ? `${pathname}#${anchor}` : pathname;
    if (`${window.location.pathname}${window.location.hash}` !== destination) window.history.pushState({}, "", destination);
    setPage(target);
    if (!anchor) {
      window.scrollTo({ top: 0, behavior: "auto" });
      return;
    }
    window.requestAnimationFrame(() => {
      window.requestAnimationFrame(() => {
        document.getElementById(anchor)?.scrollIntoView({ block: "start", behavior: "auto" });
      });
    });
  };

  const changeLanguage = (next) => {
    if (next === language) return;
    window.location.href = pathFor(next, page);
  };

  return (
    <main className="xp-page">
      <TopNav page={page} copy={copy} language={language} onNavigate={navigate} onOpenContact={() => setContactOpen(true)} />
      <ServiceHero copy={copy} language={language} page={page} onNavigate={navigate} />
      <AnimatedRoute
        key={page}
        page={page}
        copy={copy}
        language={language}
        onNavigate={navigate}
        onOpenContact={() => setContactOpen(true)}
      />
      <Footer language={language} />
      <LanguageMenu language={language} onChange={changeLanguage} />
      <ContactModal open={contactOpen} copy={copy.modal} onClose={() => setContactOpen(false)} />
    </main>
  );
}
```

```js
const source = await readFile(new URL("../src/Experience.jsx", import.meta.url), "utf8");

function blockFor(lang) {
  const start = source.indexOf(`\n  ${lang}: {`);
  assert.notEqual(start, -1, `${lang} 블록이 없습니다`);
  let depth = 0;
  for (let i = start; i < source.length; i++) {
    if (source[i] === "{") depth++;
    if (source[i] === "}") {
      depth--;
      if (depth === 0) return source.slice(start, i + 1);
    }
  }
  throw new Error(`${lang} 블록이 닫히지 않았습니다`);
}

function topKeys(block) {
  return [...block.matchAll(/^    ([a-zA-Z]+):/gm)].map((m) => m[1]).sort();
}
```

```js
  return [...].join("\n");
```

```js
#!/usr/bin/env node
import { mkdir, writeFile, readdir } from "node:fs/promises";
import path from "node:path";
import { preview } from "vite";
import { chromium } from "playwright";
import { buildHead } from "./head-template.mjs";
import { pathFor } from "../src/routing.js";

const TARGETS = [
  { lang: "ko", page: "plurank", out: "index.html" },
  { lang: "ko", page: "consulting", out: "consulting.html" },
  { lang: "en", page: "plurank", out: "en/index.html" },
  { lang: "en", page: "consulting", out: "en/consulting.html" },
  { lang: "ja", page: "plurank", out: "ja/index.html" },
  { lang: "ja", page: "consulting", out: "ja/consulting.html" },
];

const root = path.resolve(import.meta.dirname, "..");
const outDir = path.join(root, "dist", "prerendered");

const assetFiles = await readdir(path.join(root, "dist", "client", "assets"));
const assets = {
  js: "/assets/" + assetFiles.find((f) => /^index-.*\.js$/.test(f)),
  css: "/assets/" + assetFiles.find((f) => /^index-.*\.css$/.test(f)),
};

const server = await preview({ preview: { port: 4188, strictPort: true } });
const browser = await chromium.launch();
const page = await browser.newPage({ viewport: { width: 1440, height: 960 } });

for (const target of TARGETS) {
  const url = `http://localhost:4188${pathFor(target.lang, target.page)}`;
  await page.goto(url, { waitUntil: "networkidle" });
  await page.waitForSelector("#root .xp-page");
  const markup = await page.$eval("#root", (el) => el.innerHTML);

  const html = [
    "<!DOCTYPE html>",
    `<html lang="${target.lang}"><head>`,
    buildHead({ ...target, assets }),
    "</head>",
    `<body><div id="root">${markup}</div></body>`,
    "</html>",
  ].join("\n");

  const file = path.join(outDir, target.out);
  await mkdir(path.dirname(file), { recursive: true });
  await writeFile(file, html, "utf8");
  console.log("prerendered", target.out);
}

await browser.close();
await server.close();
```

```js
import assert from "node:assert/strict";
import { readFile } from "node:fs/promises";
import test from "node:test";

const PAGES = [
  { file: "index.html", lang: "ko", url: "https://www.plurank.com/" },
  { file: "consulting.html", lang: "ko", url: "https://www.plurank.com/consulting" },
  { file: "en/index.html", lang: "en", url: "https://www.plurank.com/en" },
  { file: "en/consulting.html", lang: "en", url: "https://www.plurank.com/en/consulting" },
  { file: "ja/index.html", lang: "ja", url: "https://www.plurank.com/ja" },
  { file: "ja/consulting.html", lang: "ja", url: "https://www.plurank.com/ja/consulting" },
];

const html = Object.fromEntries(
  await Promise.all(PAGES.map(async (p) => [p.file, await readFile(new URL(`../${p.file}`, import.meta.url), "utf8")])),
);

for (const page of PAGES) {
  const doc = html[page.file];

  test(`${page.file}: html lang이 ${page.lang}`, () => {
    assert.match(doc, new RegExp(`<html lang="${page.lang}"`));
  });

  test(`${page.file}: canonical이 자기 URL`, () => {
    assert.match(doc, new RegExp(`<link rel="canonical" href="${page.url.replace(/[/.]/g, "\\$&")}">`));
  });

  test(`${page.file}: hreflang 4종`, () => {
    for (const tag of ["ko", "en", "ja", "x-default"]) {
      assert.match(doc, new RegExp(`hreflang="${tag}"`));
    }
  });

  test(`${page.file}: 스냅샷 잔여물 없음`, () => {
    assert.equal(/localhost:4188/.test(doc), false);
  });
}

test("영문 페이지에 한국어가 없다", () => {
  assert.equal(/[가-힣]/.test(html["en/index.html"]), false);
  assert.equal(/[가-힣]/.test(html["en/consulting.html"]), false);
});

test("일본어 페이지에 한국어가 없다", () => {
  assert.equal(/[가-힣]/.test(html["ja/index.html"]), false);
  assert.equal(/[가-힣]/.test(html["ja/consulting.html"]), false);
});

  assert.match(html["index.html"], /AI 답변 노출을 위한/);
  assert.match(html["en/index.html"], /Multi-channel publishing/);
  assert.match(html["ja/index.html"], /マルチチャネルコンテンツ/);
});
```
